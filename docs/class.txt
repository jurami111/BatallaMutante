Model Layer ---------
interface Power:
    +getName(): String
    +getDamage(): int
    +increaseDamage(): void

interface Mutant:
    +getEnergy(): int
    +getDefense(): int
    +getPower(): Power
    +isAlive(): boolean
    +receiveDamage(int): void
    +increasePower(): void

abstract class BaseMutant implements Mutant:
    -energy: int
    -defense: int
    -power: Power
    -alive: boolean
    -position: Position
    -team: Team

class SuperHuman extends BaseMutant
class Archer extends BaseMutant
class Monster extends BaseMutant
class Dragon extends BaseMutant
class Heterodactyl extends BaseMutant

class WindBarrier implements Power
class FlameThrower implements Power
class BowShot implements Power
class SwordStrike implements Power
class DefenseBlow implements Power

Power <|.. WindBarrier
Power <|.. FlameThrower
Power <|.. BowShot
Power <|.. SwordStrike
Power <|.. DefenseBlow
Power <|.. Evade
BaseMutant o-- Power

Game Layer --------------------
class Battlefield:
    -maxTeamSize: int
    -team1: Team
    -team2: Team
    -width: int
    -height: int
    +isFinished(): boolean
    +getWinner(): Team

class Team:
    -color: Color
    -symbol: char
    -mutants: List<Mutant>
    +getAliveCount(): int
    +getDeadCount(): int

Battlefield *-- Team
Team o-- Mutant

Control Layer ---------------
class Position:
    -x: int
    -y: int
    +distanceTo(Position): double

class Movement:
    -position: Position
    -speed: int
    +moveWithin(int, int): void

enum CombatDecision {
    ATTACK
    DEFEND
}

class Combat:
    -detectionRadius: int
    +resolve(Mutant, Mutant): void
    +areInRange(Mutant, Mutant): boolean

class MutantThread implements Runnable:
    -mutant: Mutant
    -movement: Movement
    -running: boolean
    +run(): void
    +stop(): void

Mutant o-- Position
MutantThread o-- Mutant
MutantThread o-- Movement
Combat ..> Mutant

UI Layer ---------------------
interface BattleObserver:
    +onBattleUpdated(BattleSnapshot): void

class BattleSnapshot:
    +getMutants(): List<Mutant>
    +getTeamScores(): Map
    +getWinner(): Team

class GameController:
    -battlefield: Battlefield
    -battleRunning: boolean
    +startBattle(int): void
    +restartBattle(): void
    +stopBattle(): void

class GameCanvas implements BattleObserver:
    -snapshot: BattleSnapshot
    -showCombatRadius: boolean
    +onBattleUpdated(BattleSnapshot): void
    +paint(): void

class ScoreboardPanel implements BattleObserver:
    -teamAScoreLabel: JLabel
    -teamBScoreLabel: JLabel
    -winnerAnnouncementLabel: JLabel
    +onBattleUpdated(BattleSnapshot): void

class GameFrame:
    -canvasPanel: GameCanvas
    -scoreboardPanel: ScoreboardPanel
    -renderTimer: Timer
    +restart(): void

GameController o-- Battlefield
GameController --> BattleObserver
GameCanvas ..|> BattleObserver
ScoreboardPanel ..|> BattleObserver
GameFrame *-- GameCanvas
GameFrame *-- ScoreboardPanel
GameFrame --> GameController
BattleObserver ..> BattleSnapshot
